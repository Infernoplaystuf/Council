# ============================================================
#  Council AI — Quick Install
#  Pick ONE of the sections below based on your hardware, then run
#  every line in that section in order. Each section is self-contained.
#
#  WHY MULTIPLE SECTIONS:
#  - Different NVIDIA architectures need different PyTorch CUDA wheels.
#  - llama-cpp-python (the GGUF runtime) defaults to a CPU-only build
#    when installed bare from PyPI. Each GPU section pulls the matching
#    CUDA-enabled wheel via --extra-index-url so the LLM actually runs
#    on your GPU.
# ============================================================


# ============================================================
#  SECTION A — CPU only (no NVIDIA GPU)
#  Llama-cpp inference falls back to CPU; works but expect several
#  seconds per response on a 7B/8B model. Use a 3B GGUF for speed.
# ============================================================
conda config --add channels conda-forge && conda config --set channel_priority strict
conda create -n council python=3.12 -y && conda activate council
conda install -c bluequartzsoftware dream3dnx -y
pip install torch torchvision torchaudio --index-url https://download.pytorch.org/whl/cpu
pip install llama-cpp-python
pip install beautifulsoup4 chromadb crawl4ai duckdb faster-whisper h5py html2text huggingface_hub kaleido langchain-core langgraph matplotlib mido music21 numpy openpyxl pandas paramiko plotly pyarrow pymongo pypdf python-docx PyYAML requests scikit-learn scipy sentence-transformers sounddevice soundfile SQLAlchemy tkinterweb
pip install spyder-kernels && crawl4ai-setup
python -c "import torch, simplnx, llama_cpp, chromadb, sentence_transformers, pandas, openpyxl, duckdb, h5py; print('CUDA:', torch.cuda.is_available()); print('all imports OK')"

# ============================================================
#  SECTION D — NVIDIA Blackwell / RTX 50-series  (PyTorch nightly cu128)
#  Cards in this group:
#    RTX 50-series:   5070, 5070 Ti, 5080, 5090
#    Workstation:     RTX PRO 6000 Blackwell
#
#  Why a separate section: stable PyTorch's cu124 wheel was compiled
#  before Blackwell's sm_120 compute capability existed. It runs on
#  these cards only via slower PTX fallback. Native sm_120 kernels
#  ship in PyTorch's cu128 builds (stable 2.7+ or nightly).
#
#  If a stable cu128 wheel isn't available yet on your install date,
#  swap the torch line for the nightly URL; everything else stays.
# ============================================================
conda config --add channels conda-forge && conda config --set channel_priority strict
conda create -n council python=3.12 -y && conda activate council
conda install -c bluequartzsoftware dream3dnx -y
pip install --pre torch torchvision torchaudio --index-url https://download.pytorch.org/whl/nightly/cu128
pip install llama-cpp-python --extra-index-url https://abetlen.github.io/llama-cpp-python/whl/cu124 --force-reinstall --no-cache-dir
pip install beautifulsoup4 chromadb crawl4ai duckdb faster-whisper h5py html2text huggingface_hub kaleido langchain-core langgraph matplotlib mido music21 numpy openpyxl pandas paramiko plotly pyarrow pymongo pypdf python-docx PyYAML requests scikit-learn scipy sentence-transformers sounddevice soundfile SQLAlchemy tkinterweb
pip install spyder-kernels && crawl4ai-setup
python -c "import torch, simplnx, llama_cpp, chromadb, sentence_transformers, pandas, openpyxl, duckdb, h5py; print('CUDA:', torch.cuda.is_available()); print(torch.cuda.get_device_name(0) if torch.cuda.is_available() else 'CPU'); print('all imports OK')"
