# ============================================================
#  Council AI — Desktop Install Reference
#  Hardware: RTX 5080 (16 GB VRAM) / 32 GB+ RAM
#  vs Laptop build (RTX 4070 8GB): larger models, bigger context,
#  higher token limits, shorter timeouts, 2 models in VRAM at once
# ============================================================

# 1. Create environment (Python 3.11 for best torch/CUDA compatibility)
conda create -n council python=3.11 -y
conda activate council

# 2. PyTorch FIRST — must be installed before sentence-transformers
#    CUDA 12.1 is appropriate for RTX 5080 (Blackwell, sm_120)
#    If Blackwell wheels aren't yet on cu121 use cu124 or the nightly:
pip install torch torchvision torchaudio --index-url https://download.pytorch.org/whl/cu124

# 3. All required packages in one command
pip install beautifulsoup4 chromadb crawl4ai html2text kaleido langchain-core langchain-ollama langgraph matplotlib mido music21 numpy openpyxl pandas paramiko plotly requests scikit-learn scipy sentence-transformers

# 4. Optional packages
pip install tkinterweb faster-whisper sounddevice soundfile

# 5. Crawl4AI browser setup (downloads Chromium ~150 MB)
crawl4ai-setup

# 6. Pull Ollama models  (14B Q4_K_M ~9 GB each; two fit in 16 GB with OLLAMA_MAX_LOADED_MODELS=2)
ollama pull qwen2.5:14b-instruct-q4_K_M
ollama pull qwen2.5-coder:14b-instruct-q4_K_M
ollama pull phi4

# 7. Verify GPU and packages
python -c "import torch; print('CUDA:', torch.cuda.is_available(), torch.cuda.get_device_name(0) if torch.cuda.is_available() else 'NO GPU')"
python -c "
import importlib
for p in ['torch','pandas','numpy','scipy','sklearn','matplotlib','plotly','chromadb','sentence_transformers','mido','music21','langgraph','langchain_ollama','crawl4ai','bs4','html2text','tkinterweb']:
    try: importlib.import_module(p); print(f'  OK  {p}')
    except Exception as e: print(f'  MISSING  {p}: {e}')
"

# ============================================================
#  KEY DIFFERENCES vs LAPTOP BUILD (RTX 4070 8 GB)
# ============================================================
#
#  Model tier:
#    Laptop : qwen2.5:7b-instruct-q4_K_M   (~4.7 GB)
#    Desktop: qwen2.5:14b-instruct-q4_K_M  (~9.0 GB)  ← 2x parameters
#
#  Fast/judge/peasant model:
#    Laptop : phi3.5
#    Desktop: phi4                          ← newer, stronger reasoning
#
#  Context window (num_ctx):
#    Laptop : 4096 tokens
#    Desktop: 8192 tokens                  ← double the working memory
#
#  Max tokens per response:
#    general_primary: 1024  → 1800
#    coder_primary:   1200  → 2200
#    judge:            700  → 1200
#    peasant:          700  → 1200
#    default PM:      1400  → 2000
#
#  num_keep (system prompt resident in VRAM):
#    Laptop : 64 tokens
#    Desktop: 128 tokens
#
#  Timeouts:
#    Laptop : blocking=360s, streaming=420s  (cold load ~60-90s)
#    Desktop: blocking=120s, streaming=150s  (5080 loads in <5s)
#
#  Conversation history loaded per turn:
#    Laptop : 6 turns
#    Desktop: 12 turns
#
#  Concurrent models in VRAM:
#    Laptop : OLLAMA_MAX_LOADED_MODELS=1   (swap between roles)
#    Desktop: OLLAMA_MAX_LOADED_MODELS=2   (writer + coder simultaneously)
#
#  CUDA PyTorch wheel:
#    Laptop : cu121
#    Desktop: cu124  (Blackwell sm_120 support)
# ============================================================
