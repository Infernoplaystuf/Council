# ============================================================
#  Council AI — Install Reference (GGUF-only build)
#
#  The council loads a single Hugging Face .gguf model from disk via
#  llama-cpp-python. Ollama is no longer used. Designed for air-gapped
#  machines where the user transfers a .gguf onto disk and runs the
#  app fully offline.
# ============================================================

# ─────────────────────────────────────────────────────────────
#  STEP 1 — Conda environment
# ─────────────────────────────────────────────────────────────
# Python 3.11 is required for torch + llama-cpp wheel compatibility.
# `spyder-kernels=2.5` is only needed if you'll run the app from Spyder.
conda create -n council python=3.11 spyder-kernels=2.5 -c conda-forge -y
conda activate council

# ─────────────────────────────────────────────────────────────
#  STEP 2 — PyTorch (install FIRST, before sentence-transformers)
# ─────────────────────────────────────────────────────────────
#   Desktop / RTX 5080 (Blackwell sm_120): use cu124
#   Laptop  / RTX 4070  (Ada,  sm_89):    use cu121
#   CPU-only machines:                    use --index-url .../cpu
pip install torch torchvision torchaudio --index-url https://download.pytorch.org/whl/cu124
# (CPU fallback)
# pip install torch torchvision torchaudio --index-url https://download.pytorch.org/whl/cpu

# ─────────────────────────────────────────────────────────────
#  STEP 3 — Core dependencies
# ─────────────────────────────────────────────────────────────
pip install \
    beautifulsoup4 \
    chromadb \
    crawl4ai \
    html2text \
    kaleido \
    langchain-core \
    langgraph \
    matplotlib \
    mido \
    music21 \
    numpy \
    openpyxl \
    pandas \
    paramiko \
    plotly \
    PyYAML \
    requests \
    scikit-learn \
    scipy \
    sentence-transformers

# ─────────────────────────────────────────────────────────────
#  STEP 4 — File parsing helpers (for the vault analyst over PDFs/DOCX)
# ─────────────────────────────────────────────────────────────
conda install -c conda-forge pypdf python-docx huggingface_hub -y

# ─────────────────────────────────────────────────────────────
#  STEP 5 — Optional packages
# ─────────────────────────────────────────────────────────────
pip install tkinterweb faster-whisper sounddevice soundfile

# ─────────────────────────────────────────────────────────────
#  STEP 6 — Crawl4AI browser setup (downloads Chromium ~150 MB; skip if offline)
# ─────────────────────────────────────────────────────────────
crawl4ai-setup

# ─────────────────────────────────────────────────────────────
#  STEP 7 — Local GGUF inference backend (llama-cpp-python)
# ─────────────────────────────────────────────────────────────
# CPU-only build:
conda install -c conda-forge llama-cpp-python -y
#
# GPU build (CUDA — much faster):
# pip install llama-cpp-python --extra-index-url https://abetlen.github.io/llama-cpp-python/whl/cu124

# ─────────────────────────────────────────────────────────────
#  STEP 8 — Download a GGUF model
# ─────────────────────────────────────────────────────────────
# Recommended US-licensed picks:
#   Laptop  (8 GB VRAM):  bartowski/granite-3.0-8b-instruct-GGUF (Q4_K_M)
#   Desktop (16 GB VRAM): bartowski/phi-4-GGUF (Q4_K_M)
#
# One-time download on a machine with internet:
# huggingface-cli download bartowski/granite-3.0-8b-instruct-GGUF granite-3.0-8b-instruct-Q4_K_M.gguf --local-dir C:\models
#
# Transfer the .gguf to your offline machine, then open the app and
# click "Browse..." in the GGUF model picker at the top of the Council tab.

# ─────────────────────────────────────────────────────────────
#  STEP 9 — Optional environment variable overrides
# ─────────────────────────────────────────────────────────────
# These can be set instead of (or in addition to) the in-app picker:
#   set COUNCIL_GGUF_PATH=C:\models\granite-3.0-8b-instruct-Q4_K_M.gguf
#   set COUNCIL_GGUF_N_CTX=4096          (context window)
#   set COUNCIL_GGUF_N_THREADS=8         (CPU threads; defaults to all cores)
#   set COUNCIL_GGUF_GPU_LAYERS=99       (0 = CPU only, 99 = all on GPU)
# (PowerShell: use $env:NAME="value")

# ─────────────────────────────────────────────────────────────
#  STEP 10 — Verify install
# ─────────────────────────────────────────────────────────────
python -c "import torch; print('CUDA:', torch.cuda.is_available(), torch.cuda.get_device_name(0) if torch.cuda.is_available() else 'CPU only')"
python -c "
import importlib
for p in ['torch','pandas','numpy','scipy','sklearn','matplotlib','plotly',
          'chromadb','sentence_transformers','mido','music21','langgraph',
          'crawl4ai','bs4','html2text','tkinterweb',
          'pypdf','docx','huggingface_hub','llama_cpp']:
    try: importlib.import_module(p); print(f'  OK       {p}')
    except Exception as e: print(f'  MISSING  {p}: {e}')
"

# ============================================================
#  Hardware tier guidance
# ============================================================
#  Laptop (RTX 4070, 8 GB VRAM):
#    - 7-8B Q4_K_M GGUF fits with 4-8K context
#    - COUNCIL_GGUF_N_CTX=4096, COUNCIL_GGUF_GPU_LAYERS=99
#
#  Desktop (RTX 5080, 16 GB VRAM):
#    - 14B Q4_K_M GGUF fits with 8K context (Phi-4 is a strong pick)
#    - COUNCIL_GGUF_N_CTX=8192, COUNCIL_GGUF_GPU_LAYERS=99
#
#  CPU-only:
#    - Use a 7B Q4 model; set COUNCIL_GGUF_GPU_LAYERS=0
#    - Set COUNCIL_GGUF_N_THREADS to your core count
# ============================================================
