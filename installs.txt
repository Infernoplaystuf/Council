# ============================================================
#  Council AI — Install Reference
#  Two install paths:
#    A) Ollama backend (default, easiest, faster on GPU)
#    B) Direct GGUF backend (air-gapped / no Ollama daemon)
#  You can install both and switch at runtime via COUNCIL_BACKEND.
# ============================================================

# ─────────────────────────────────────────────────────────────
#  STEP 1 — Conda environment
# ─────────────────────────────────────────────────────────────
# Python 3.11 is required for torch/CUDA + llama-cpp wheel compatibility.
# `spyder-kernels=2.5` makes the env usable from Spyder's IPython console
# (skip it if you're not using Spyder).
conda create -n council python=3.11 spyder-kernels=2.5 -c conda-forge -y
conda activate council

# ─────────────────────────────────────────────────────────────
#  STEP 2 — PyTorch (install FIRST, before sentence-transformers)
# ─────────────────────────────────────────────────────────────
#   Desktop / RTX 5080 (Blackwell sm_120): use cu124
#   Laptop  / RTX 4070  (Ada,  sm_89):    use cu121
#   CPU-only machines:                    use --index-url .../cpu
pip install torch torchvision torchaudio --index-url https://download.pytorch.org/whl/cu124
# (CPU fallback)
# pip install torch torchvision torchaudio --index-url https://download.pytorch.org/whl/cpu

# ─────────────────────────────────────────────────────────────
#  STEP 3 — Core dependencies (always required)
# ─────────────────────────────────────────────────────────────
pip install \
    beautifulsoup4 \
    chromadb \
    crawl4ai \
    html2text \
    kaleido \
    langchain-core \
    langchain-ollama \
    langgraph \
    matplotlib \
    mido \
    music21 \
    numpy \
    openpyxl \
    pandas \
    paramiko \
    plotly \
    PyYAML \
    requests \
    scikit-learn \
    scipy \
    sentence-transformers

# ─────────────────────────────────────────────────────────────
#  STEP 4 — File parsing helpers (for vault analyst / RAG over PDFs/DOCX)
# ─────────────────────────────────────────────────────────────
conda install -c conda-forge pypdf python-docx huggingface_hub -y

# ─────────────────────────────────────────────────────────────
#  STEP 5 — Optional packages
# ─────────────────────────────────────────────────────────────
pip install tkinterweb faster-whisper sounddevice soundfile

# ─────────────────────────────────────────────────────────────
#  STEP 6 — Crawl4AI browser setup (downloads Chromium ~150 MB)
# ─────────────────────────────────────────────────────────────
crawl4ai-setup

# ─────────────────────────────────────────────────────────────
#  STEP 7 — Backend A: Ollama
# ─────────────────────────────────────────────────────────────
# Install Ollama from https://ollama.com/download
# Then pull the models you want:
ollama pull qwen2.5:14b-instruct-q4_K_M
ollama pull qwen2.5-coder:14b-instruct-q4_K_M
ollama pull phi4
# (laptop / 8GB VRAM users — pull 7b instead:)
# ollama pull qwen2.5:7b-instruct-q4_K_M
# ollama pull qwen2.5-coder:7b-instruct-q4_K_M
# ollama pull phi3.5

# ─────────────────────────────────────────────────────────────
#  STEP 8 — Backend B: Direct GGUF (no Ollama; works offline / air-gapped)
# ─────────────────────────────────────────────────────────────
# `llama-cpp-python` lets the council load a .gguf file directly so you can:
#   - Transfer a GGUF onto an offline machine and just point at it
#   - Use any Hugging Face GGUF without going through an Ollama Modelfile
#
# CPU-only build:
conda install -c conda-forge llama-cpp-python -y
#
# GPU build (CUDA — much faster generation; requires CUDA toolkit installed):
# pip install llama-cpp-python --extra-index-url https://abetlen.github.io/llama-cpp-python/whl/cu124
#
# Downloading a GGUF from Hugging Face (one-time, on a machine WITH internet):
# huggingface-cli download Qwen/Qwen2.5-7B-Instruct-GGUF qwen2.5-7b-instruct-q4_k_m.gguf --local-dir ./models
# Then copy the .gguf to your offline machine.

# ─────────────────────────────────────────────────────────────
#  STEP 9 — Choose the backend at runtime (env vars)
# ─────────────────────────────────────────────────────────────
# Default (Ollama) — no env vars needed.
#
# To use a local GGUF instead:
#   set COUNCIL_BACKEND=gguf
#   set COUNCIL_GGUF_PATH=C:\path\to\model.gguf
# Optional GGUF tuning:
#   set COUNCIL_GGUF_N_CTX=4096          (context window)
#   set COUNCIL_GGUF_N_THREADS=8         (CPU threads; defaults to all cores)
#   set COUNCIL_GGUF_GPU_LAYERS=0        (0 = CPU only, 99 = all on GPU)
#
# (PowerShell uses $env:NAME="value" instead of set NAME=value)

# ─────────────────────────────────────────────────────────────
#  STEP 10 — Verify install
# ─────────────────────────────────────────────────────────────
python -c "import torch; print('CUDA:', torch.cuda.is_available(), torch.cuda.get_device_name(0) if torch.cuda.is_available() else 'CPU only')"
python -c "
import importlib
for p in ['torch','pandas','numpy','scipy','sklearn','matplotlib','plotly',
          'chromadb','sentence_transformers','mido','music21','langgraph',
          'langchain_ollama','crawl4ai','bs4','html2text','tkinterweb',
          'pypdf','docx','huggingface_hub','llama_cpp']:
    try: importlib.import_module(p); print(f'  OK       {p}')
    except Exception as e: print(f'  MISSING  {p}: {e}')
"
